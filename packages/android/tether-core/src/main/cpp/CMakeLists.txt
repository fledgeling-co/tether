cmake_minimum_required(VERSION 3.22.1)
project(tether_jni C)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(TETHER_TRIPLE aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(TETHER_TRIPLE x86_64-linux-android)
else()
  message(FATAL_ERROR "Tether supports arm64-v8a and x86_64; build another Rust target explicitly before adding an ABI")
endif()
set(TETHER_ARCHIVE "${TETHER_RUST_DIR}/${TETHER_TRIPLE}/release/libtether_ffi.a")
if(NOT EXISTS "${TETHER_ARCHIVE}")
  message(FATAL_ERROR "Run scripts/build-android.sh before building the Android library")
endif()
add_library(tether_ffi STATIC IMPORTED)
set_target_properties(tether_ffi PROPERTIES IMPORTED_LOCATION "${TETHER_ARCHIVE}")
add_library(tether_jni SHARED tether_jni.c)
target_link_libraries(tether_jni tether_ffi log android dl m)
target_compile_options(tether_jni PRIVATE -Wall -Wextra -Werror -fvisibility=hidden "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/tether/source/android/jni")
target_link_options(tether_jni PRIVATE "-Wl,-z,max-page-size=16384")
